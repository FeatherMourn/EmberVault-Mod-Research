-- FeatherMourn: Item Selector / single-use inventory slot editor
-- Catalog data: supplied reference item list (names and hashes retained unchanged).
-- Original UI / safeguards rewritten; catalog identity is NOT an independent discovery.
-- Requires an independently VERIFIED pointer capture named InventoryItemSlot,
-- or the existing working capture named pItem. Neither is installed here.
-- Build reported: 1024333. Item layout at +0 and +8 remains experimental.
-- Run only in a backed-up offline test world; never enable two competing hooks.

local POINTER_SYMBOLS = { 'InventoryItemSlot', 'pItem' }
local function unsigned32(n)
  return n % 4294967296
end

local function findCapturedItem()
  local descriptions = {}
  for _, symbol in ipairs(POINTER_SYMBOLS) do
    local slot = getAddressSafe(symbol)
    if slot and slot ~= 0 then
      local item = readQword(slot)
      if item and item ~= 0 then
        local oldHash = readInteger(item)
        local oldCount = readInteger(item + 8)
        if oldHash ~= nil and oldCount ~= nil then
          return item, slot, symbol, oldHash, oldCount
        end
        descriptions[#descriptions + 1] = symbol .. ': pointer unreadable'
      else
        descriptions[#descriptions + 1] = symbol .. ': no item captured'
      end
    else
      descriptions[#descriptions + 1] = symbol .. ': symbol missing'
    end
  end
  showMessage('No usable inventory item pointer. Enable ONE known-working '
    .. 'inventory capture, select a disposable inventory item, and retry.\n'
    .. table.concat(descriptions, '\n'))
  return nil
end

local function buildCatalog()
  local result = {}
  for line in catalogText:gmatch('[^\r\n]+') do
    local name, hex = line:match('^(.-)\t(%x%x%x%x%x%x%x%x)$')
    if not name or not hex then
      showMessage('Invalid catalog row: ' .. line)
      return nil
    end
    result[#result + 1] = { name = name, hash = tonumber(hex, 16), hex = hex }
  end
  table.sort(result, function(a, b)
    local al, bl = a.name:lower(), b.name:lower()
    if al == bl then return a.hex < b.hex end
    return al < bl
  end)
  return result
end

local function pickItem(catalog)
  local search = inputQuery('FeatherMourn Item Selector',
    'Enter part of the item name (leave blank to show every item):', '')
  if search == nil then return nil end
  search = search:lower()
  local matches = {}
  local options = createStringList()
  for _, row in ipairs(catalog) do
    if search == '' or row.name:lower():find(search, 1, true) then
      matches[#matches + 1] = row
      options.add(string.format('%s   [0x%s]', row.name, row.hex))
    end
  end
  if #matches == 0 then
    options.destroy()
    showMessage('No item name matched: ' .. search)
    return nil
  end
  local index = showSelectionList('Select inventory item',
    'Choose one of ' .. tostring(#matches) .. ' results:', options, false)
  options.destroy()
  if index == nil or index < 0 then return nil end
  return matches[index + 1]
end

local item, pointerSlot, source, originalHash, originalCount = findCapturedItem()
if item then
  local catalog = buildCatalog()
  if catalog then
    local selected = pickItem(catalog)
    if selected then
      local answer = messageDialog(
        string.format('Pointer source: %s\nItem address: %X\n'
          .. 'Old hash: 0x%08X\nOld value at +8: %d\n\n'
          .. 'Replace selected slot with:\n%s (0x%s)\n'
          .. 'Set +8 to 1 as in the supplied reference script?\n\n'
          .. 'Only confirm on a backed-up offline test world.',
          source, item, unsigned32(originalHash), originalCount,
          selected.name, selected.hex),
        mtConfirmation, mbYes, mbNo)
      if answer == mrYes then
        -- Fail closed if the selection or either presumed field has changed.
        local current = readQword(pointerSlot)
        if current ~= item or readInteger(item) ~= originalHash
          or readInteger(item + 8) ~= originalCount then
          showMessage('Inventory slot changed while the selector was open. '
            .. 'No changes made. Reopen the selector after capturing the item.')
        elseif not writeInteger(item, selected.hash) then
          showMessage('Item hash write failed; count not changed.')
        elseif not writeInteger(item + 8, 1) then
          local restored = writeInteger(item, originalHash)
          showMessage('Value at +8 could not be written. '
            .. (restored and 'Original hash restored.'
              or 'Hash restoration failed; inspect slot before saving!'))
        elseif readInteger(item) == nil or
          unsigned32(readInteger(item)) ~= selected.hash or
          readInteger(item + 8) ~= 1 then
          -- Best-effort rollback on verification failure.
          local r1 = writeInteger(item, originalHash)
          local r2 = writeInteger(item + 8, originalCount)
          showMessage('Verification failed. Rollback: '
            .. ((r1 and r2) and 'both writes attempted successfully.'
              or 'INCOMPLETE: inspect item slot before saving!'))
        else
          showMessage('Memory write verified for ' .. selected.name
            .. '. In-game item behavior is NOT yet validated.')
        end
      end
    end
  end
end
